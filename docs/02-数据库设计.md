# 数据库设计

## 1. 分层策略

| 数据层 | 存储方案 |
|--------|---------|
| 业务主数据 | PostgreSQL |
| 事件数据 | PostgreSQL (partition by task_id hash) |
| 证据对象 | 本地文件系统 + PostgreSQL元数据 |
| 缓存与队列 | Redis |
| 审计数据 | PostgreSQL 独立表 |

## 2. ER关系图

```
Project 1──N App 1──N AppVersion 1──N DetectionTask
DetectionTask 1──N SubTask
DetectionTask 1──N DetectionScenario 1──N DetectionEvent
DetectionEvent N──N Evidence (via event_evidence)
RuleVersion 1──N Rule 1──N Finding
Finding N──N DetectionEvent / Evidence
AppVersion N──N SDKKnowledge (via version_sdk)
Finding 1──N Remediation 1──N RetestRecord
```

## 3. 核心数据表定义

### 3.1 组织与认证

```sql
-- 用户表
CREATE TABLE users (
    id BIGSERIAL PRIMARY KEY,
    username VARCHAR(100) UNIQUE NOT NULL,
    email VARCHAR(255) UNIQUE,
    full_name VARCHAR(100),
    phone VARCHAR(20),
    password_hash VARCHAR(255) NOT NULL,
    role VARCHAR(50) NOT NULL DEFAULT 'viewer', -- platform_admin/rule_admin/project_owner/tester/developer/compliance/auditor
    status VARCHAR(20) NOT NULL DEFAULT 'active', -- active/disabled
    last_login_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 项目表
CREATE TABLE projects (
    id BIGSERIAL PRIMARY KEY,
    name VARCHAR(200) NOT NULL,
    description TEXT,
    owner_id BIGINT REFERENCES users(id),
    status VARCHAR(20) NOT NULL DEFAULT 'active',
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 项目成员表
CREATE TABLE project_members (
    id BIGSERIAL PRIMARY KEY,
    project_id BIGINT NOT NULL REFERENCES projects(id) ON DELETE CASCADE,
    user_id BIGINT NOT NULL REFERENCES users(id),
    role VARCHAR(50) NOT NULL DEFAULT 'tester', -- owner/tester/developer/compliance
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    UNIQUE(project_id, user_id)
);
```

### 3.2 App资产

```sql
-- App资产表
CREATE TABLE app_assets (
    id BIGSERIAL PRIMARY KEY,
    project_id BIGINT NOT NULL REFERENCES projects(id) ON DELETE CASCADE,
    package_name VARCHAR(255) NOT NULL,
    app_name VARCHAR(200) NOT NULL,
    app_type VARCHAR(50) DEFAULT 'android',
    category VARCHAR(100),
    department VARCHAR(200),
    vendor VARCHAR(200),
    owner_id BIGINT REFERENCES users(id),
    description TEXT,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    UNIQUE(project_id, package_name)
);

-- App版本表
CREATE TABLE app_versions (
    id BIGSERIAL PRIMARY KEY,
    app_id BIGINT NOT NULL REFERENCES app_assets(id) ON DELETE CASCADE,
    version_name VARCHAR(100) NOT NULL,
    version_code INTEGER NOT NULL,
    sha256 VARCHAR(64) NOT NULL,
    file_size BIGINT,
    min_sdk INTEGER,
    target_sdk INTEGER,
    signature_info JSONB,
    artifact_path VARCHAR(500),
    channel VARCHAR(100),
    build_time TIMESTAMPTZ,
    upload_notes TEXT,
    uploaded_by BIGINT REFERENCES users(id),
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
CREATE INDEX idx_app_versions_app ON app_versions(app_id);
CREATE INDEX idx_app_versions_sha ON app_versions(sha256);

-- 隐私政策表
CREATE TABLE privacy_policies (
    id BIGSERIAL PRIMARY KEY,
    app_id BIGINT NOT NULL REFERENCES app_assets(id) ON DELETE CASCADE,
    version VARCHAR(100),
    effective_date DATE,
    raw_text TEXT,
    raw_artifact_path VARCHAR(500),
    parsed_json JSONB,
    parse_status VARCHAR(20) DEFAULT 'pending', -- pending/parsed/failed/reviewed
    parse_model VARCHAR(100),
    parse_confidence FLOAT,
    reviewed_by BIGINT REFERENCES users(id),
    reviewed_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
```

### 3.3 检测任务

```sql
-- 检测任务表
CREATE TABLE detection_tasks (
    id BIGSERIAL PRIMARY KEY,
    task_code VARCHAR(50) UNIQUE NOT NULL,
    app_version_id BIGINT NOT NULL REFERENCES app_versions(id),
    privacy_policy_id BIGINT REFERENCES privacy_policies(id),
    project_id BIGINT NOT NULL REFERENCES projects(id),
    detection_type VARCHAR(50) NOT NULL DEFAULT 'full', -- full/static_only/consent_pre/sdk_audit/version_diff
    rule_pack_version VARCHAR(50) NOT NULL,
    status VARCHAR(30) NOT NULL DEFAULT 'draft',
    -- draft/queued/preparing/running_static/waiting_dynamic/running_dynamic/analyzing/reviewing/completed/failed/canceled
    priority INTEGER DEFAULT 5,
    config_json JSONB NOT NULL DEFAULT '{}',
    -- {dynamic_scenarios, device_requirements, timeout, retry_policy, report_template}
    started_at TIMESTAMPTZ,
    completed_at TIMESTAMPTZ,
    failed_reason TEXT,
    created_by BIGINT REFERENCES users(id),
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
CREATE INDEX idx_tasks_project ON detection_tasks(project_id);
CREATE INDEX idx_tasks_status ON detection_tasks(status);
CREATE INDEX idx_tasks_app_version ON detection_tasks(app_version_id);

-- 子任务表
CREATE TABLE sub_tasks (
    id BIGSERIAL PRIMARY KEY,
    task_id BIGINT NOT NULL REFERENCES detection_tasks(id) ON DELETE CASCADE,
    sub_task_code VARCHAR(50) UNIQUE NOT NULL,
    engine_type VARCHAR(50) NOT NULL, -- mobsf/appshark/androguard/jadx/frida_agent/mitmproxy
    stage VARCHAR(50) NOT NULL, -- prepare/execute/analyze/done
    status VARCHAR(30) NOT NULL DEFAULT 'pending', -- pending/running/completed/failed/canceled
    assigned_node_id BIGINT,
    config_json JSONB,
    result_summary JSONB,
    started_at TIMESTAMPTZ,
    completed_at TIMESTAMPTZ,
    error_message TEXT,
    retry_count INTEGER DEFAULT 0,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
CREATE INDEX idx_subtasks_task ON sub_tasks(task_id);

-- 检测场景表
CREATE TABLE detection_scenarios (
    id BIGSERIAL PRIMARY KEY,
    task_id BIGINT NOT NULL REFERENCES detection_tasks(id) ON DELETE CASCADE,
    scenario_type VARCHAR(50) NOT NULL, -- first_launch/rejected/consented/function_trigger/revoked/account_cancel
    consent_status VARCHAR(30) NOT NULL, -- not_presented/not_agreed/rejected/consented/revoked
    status VARCHAR(30) NOT NULL DEFAULT 'pending', -- pending/running/completed/failed/skipped
    started_at TIMESTAMPTZ,
    completed_at TIMESTAMPTZ,
    device_id BIGINT,
    operator_id BIGINT REFERENCES users(id),
    notes TEXT,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
CREATE INDEX idx_scenarios_task ON detection_scenarios(task_id);
```

### 3.4 事件与证据

```sql
-- 统一检测事件表
CREATE TABLE detection_events (
    id BIGSERIAL PRIMARY KEY,
    event_uid VARCHAR(50) UNIQUE NOT NULL, -- evt_xxx
    task_id BIGINT NOT NULL REFERENCES detection_tasks(id) ON DELETE CASCADE,
    scenario_id BIGINT REFERENCES detection_scenarios(id),
    event_type VARCHAR(50) NOT NULL,
    -- sensitive_api_call / network_request / permission_request / page_view / static_data_flow / consent_state_change / sdk_init
    timestamp TIMESTAMPTZ NOT NULL,
    consent_status VARCHAR(30),
    data_type VARCHAR(100), -- ANDROID_ID/OAID/IMEI/MAC/LOCATION/CONTACTS/PHONE_LIST/IMSI etc
    api VARCHAR(500),
    caller VARCHAR(500),
    sdk_id BIGINT REFERENCES sdk_knowledge(id),
    trace_id VARCHAR(100),
    value_fingerprint VARCHAR(64),
    event_data JSONB NOT NULL DEFAULT '{}',
    engine_name VARCHAR(100),
    engine_version VARCHAR(50),
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
CREATE INDEX idx_events_task ON detection_events(task_id);
CREATE INDEX idx_events_scenario ON detection_events(scenario_id);
CREATE INDEX idx_events_type ON detection_events(event_type);
CREATE INDEX idx_events_timestamp ON detection_events(timestamp);
CREATE INDEX idx_events_data_type ON detection_events(data_type);

-- 证据表
CREATE TABLE evidence (
    id BIGSERIAL PRIMARY KEY,
    evidence_uid VARCHAR(50) UNIQUE NOT NULL,
    task_id BIGINT NOT NULL REFERENCES detection_tasks(id) ON DELETE CASCADE,
    evidence_type VARCHAR(50) NOT NULL,
    -- static_data_flow / api_call / network_request / screenshot / screen_recording / log_file / engine_output / human_note
    artifact_path VARCHAR(500),
    artifact_hash VARCHAR(64),
    artifact_size BIGINT,
    metadata_json JSONB NOT NULL DEFAULT '{}',
    -- {api_call_stack, network_domain, network_path, screen_page_id, etc}
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
CREATE INDEX idx_evidence_task ON evidence(task_id);

-- 事件-证据关联表
CREATE TABLE event_evidence (
    id BIGSERIAL PRIMARY KEY,
    event_id BIGINT NOT NULL REFERENCES detection_events(id) ON DELETE CASCADE,
    evidence_id BIGINT NOT NULL REFERENCES evidence(id) ON DELETE CASCADE,
    relation_type VARCHAR(50), -- primary/supporting/context
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    UNIQUE(event_id, evidence_id)
);
```

### 3.5 规则与问题

```sql
-- 规则主数据
CREATE TABLE rules (
    id BIGSERIAL PRIMARY KEY,
    rule_key VARCHAR(100) UNIQUE NOT NULL, -- PRIV-CONSENT-001
    name VARCHAR(200) NOT NULL,
    category VARCHAR(50) NOT NULL, -- consent/sdk/permission/network/policy/right/security/account
    description TEXT,
    current_version_id BIGINT,
    status VARCHAR(20) DEFAULT 'active', -- draft/testing/active/deprecated
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 规则版本表
CREATE TABLE rule_versions (
    id BIGSERIAL PRIMARY KEY,
    rule_id BIGINT NOT NULL REFERENCES rules(id) ON DELETE CASCADE,
    version VARCHAR(20) NOT NULL,
    rule_content JSONB NOT NULL, -- YAML定义转JSON
    -- {scope, when, exclude, evidence_required, severity, remediation_template}
    changelog TEXT,
    status VARCHAR(20) DEFAULT 'draft', -- draft/testing/published/deprecated
    published_by BIGINT REFERENCES users(id),
    published_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    UNIQUE(rule_id, version)
);

-- 风险问题表
CREATE TABLE findings (
    id BIGSERIAL PRIMARY KEY,
    finding_uid VARCHAR(50) UNIQUE NOT NULL,
    task_id BIGINT NOT NULL REFERENCES detection_tasks(id) ON DELETE CASCADE,
    rule_id BIGINT REFERENCES rules(id),
    rule_version_id BIGINT REFERENCES rule_versions(id),
    scenario_id BIGINT REFERENCES detection_scenarios(id),
    severity VARCHAR(20) NOT NULL, -- critical/high/medium/low/info
    status VARCHAR(30) NOT NULL DEFAULT 'open',
    -- open/assigned/fixing/ready_for_retest/retesting/closed_fixed/closed_accepted/closed_false_positive/reopened
    confidence VARCHAR(20) DEFAULT 'confirmed', -- confirmed/probable/possible
    title VARCHAR(500) NOT NULL,
    description TEXT,
    data_type VARCHAR(100),
    sdk_id BIGINT REFERENCES sdk_knowledge(id),
    network_domain VARCHAR(255),
    api_path VARCHAR(500),
    remediation_advice TEXT,
    assigned_to BIGINT REFERENCES users(id),
    assigned_at TIMESTAMPTZ,
    due_date DATE,
    closed_reason VARCHAR(50),
    closed_by BIGINT REFERENCES users(id),
    closed_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
CREATE INDEX idx_findings_task ON findings(task_id);
CREATE INDEX idx_findings_severity ON findings(severity);
CREATE INDEX idx_findings_status ON findings(status);

-- 问题-证据关联表
CREATE TABLE finding_evidence (
    id BIGSERIAL PRIMARY KEY,
    finding_id BIGINT NOT NULL REFERENCES findings(id) ON DELETE CASCADE,
    evidence_id BIGINT NOT NULL REFERENCES evidence(id) ON DELETE CASCADE,
    relation_type VARCHAR(50), -- primary/supporting
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    UNIQUE(finding_id, evidence_id)
);

-- 问题-事件关联表
CREATE TABLE finding_events (
    id BIGSERIAL PRIMARY KEY,
    finding_id BIGINT NOT NULL REFERENCES findings(id) ON DELETE CASCADE,
    event_id BIGINT NOT NULL REFERENCES detection_events(id) ON DELETE CASCADE,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    UNIQUE(finding_id, event_id)
);
```

### 3.6 SDK知识库

```sql
-- SDK知识表
CREATE TABLE sdk_knowledge (
    id BIGSERIAL PRIMARY KEY,
    sdk_uid VARCHAR(50) UNIQUE NOT NULL,
    name VARCHAR(200) NOT NULL,
    vendor VARCHAR(200),
    category VARCHAR(100), -- analytics/ads/push/map/social/payment/crash/other
    official_url VARCHAR(500),
    privacy_policy_url VARCHAR(500),
    status VARCHAR(20) DEFAULT 'active',
    privacy_behaviors JSONB DEFAULT '[]',
    -- [{data_type, apis, domains, init_timing}]
    config_capabilities JSONB DEFAULT '{}',
    -- {delay_init, opt_out, disable_device_id, exit_api}
    description TEXT,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- SDK指纹表
CREATE TABLE sdk_fingerprints (
    id BIGSERIAL PRIMARY KEY,
    sdk_id BIGINT NOT NULL REFERENCES sdk_knowledge(id) ON DELETE CASCADE,
    fingerprint_type VARCHAR(50) NOT NULL,
    -- package_name/class_name/method/file_hash/resource/native/manifest_component/domain
    fingerprint_value TEXT NOT NULL,
    weight FLOAT DEFAULT 1.0,
    version_range VARCHAR(100),
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
CREATE INDEX idx_sdk_fp_value ON sdk_fingerprints(fingerprint_value);

-- App版本-SDK关联表
CREATE TABLE version_sdks (
    id BIGSERIAL PRIMARY KEY,
    app_version_id BIGINT NOT NULL REFERENCES app_versions(id) ON DELETE CASCADE,
    sdk_id BIGINT REFERENCES sdk_knowledge(id),
    identified_at TIMESTAMPTZ DEFAULT NOW(),
    confidence VARCHAR(20) DEFAULT 'confirmed', -- confirmed/probable/possible/unknown
    match_details JSONB,
    UNIQUE(app_version_id, sdk_id)
);
```

### 3.7 整改与复测

```sql
-- 整改表
CREATE TABLE remediations (
    id BIGSERIAL PRIMARY KEY,
    finding_id BIGINT NOT NULL REFERENCES findings(id) ON DELETE CASCADE,
    remediation_type VARCHAR(50), -- code_fix/config_adjust/policy_update/accepted/false_positive
    description TEXT,
    fix_version_id BIGINT REFERENCES app_versions(id),
    attachments JSONB DEFAULT '[]',
    submitted_by BIGINT REFERENCES users(id),
    submitted_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    status VARCHAR(30) DEFAULT 'pending_retest', -- pending_retest/retest_passed/retest_failed
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 复测记录表
CREATE TABLE retest_records (
    id BIGSERIAL PRIMARY KEY,
    original_finding_id BIGINT NOT NULL REFERENCES findings(id),
    retest_task_id BIGINT REFERENCES detection_tasks(id),
    result VARCHAR(30), -- passed/failed/inconclusive
    notes TEXT,
    tested_by BIGINT REFERENCES users(id),
    tested_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
```

### 3.8 系统管理

CREATE TABLE devices (

    id BIGSERIAL PRIMARY KEY,

    dict_type VARCHAR(100) NOT NULL,

    status VARCHAR(20) DEFAULT 'active',

    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
